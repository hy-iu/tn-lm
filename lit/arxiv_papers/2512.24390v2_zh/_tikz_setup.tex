\newcommand{\middleCoo}[6]{
    \path (#1) --coordinate[pos=0.5](#3) (#2);
    \path (#1) --coordinate[pos={0.5-\sp}](#4) (#2);
    \path (#1) --coordinate[pos={0.5+\sp}](#5) (#2);
}
